import lark_oapi as lark
from lark_oapi.api.im.v1 import *
import requests
import json

# ========== 配置项 ==========
# 飞书应用信息
APP_ID = "你的飞书AppID"
APP_SECRET = "你的飞书AppSecret"

# DeepSeek 配置
DS_API_KEY = "你的DeepSeek密钥"
DS_API_URL = "https://api.deepseek.com/chat/completions"
MODEL = "deepseek-chat"
# 办公互动专用 0.3 严谨稳重
TEMP = 0.3

# 职场助手人设
SYS_PROMPT = "你是企业内部办公助手，回答正式简洁、逻辑稳妥，贴合职场沟通，不闲聊发散，务实解答工作各类问题。"

# ========== 调用DeepSeek ==========
def get_deepseek_reply(question: str) -> str:
    headers = {
        "Authorization": f"Bearer {DS_API_KEY}",
        "Content-Type": "application/json"
    }
    body = {
        "model": MODEL,
        "messages": [
            {"role": "system", "content": SYS_PROMPT},
            {"role": "user", "content": question}
        ],
        "temperature": TEMP,
        "top_p": 0.88,
        "max_tokens": 4096,
        "frequency_penalty": 0.08,
        "presence_penalty": 0.05,
        "stream": False
    }
    try:
        resp = requests.post(DS_API_URL, headers=headers, json=body, timeout=25)
        data = resp.json()
        return data["choices"][0]["message"]["content"].strip()
    except Exception as e:
        return f"AI应答异常：{str(e)}"

# ========== 监听飞书消息回调 ==========
def handle_message(event: P2ImMessageReceiveV1) -> None:
    msg = event.event.message
    # 只处理文本消息
    if msg.message_type != "text":
        return

    # 解析用户文本
    user_text = json.loads(msg.content).get("text", "")
    if not user_text.strip():
        return

    print(f"收到飞书消息：{user_text}")
    ai_reply = get_deepseek_reply(user_text)
    print(f"AI回复：{ai_reply}")

    # 构造发送客户端
    client = lark.Client.builder()\
        .app_id(APP_ID)\
        .app_secret(APP_SECRET)\
        .log_level(lark.LogLevel.INFO)\
        .build()

    # 回复原会话
    send_req = CreateMessageRequest.builder()\
        .receive_id(msg.chat_id)\
        .msg_type("text")\
        .content(json.dumps({"text": ai_reply}))\
        .build()
    client.im.v1.message.create(send_req)

# ========== 启动长连接监听 ==========
def main():
    # 注册消息事件
    dispatcher = lark.EventDispatcherHandler.builder("", "")\
        .register_p2_im_message_receive_v1(handle_message)\
        .build()

    # 建立飞书WebSocket长连接
    ws_client = lark.ws.Client(
        app_id=APP_ID,
        app_secret=APP_SECRET,
        event_handler=dispatcher,
        log_level=lark.LogLevel.INFO
    )
    print("=== 飞书长连接机器人已启动 ===")
    print("=== 已接入DeepSeek 温度0.3 办公模式 ===")
    ws_client.start()

if __name__ == "__main__":
    main()
